public class Termometro
{
    private double _temperaturaCelsius;

    public double Celsius
    {
        get { return _temperaturaCelsius; }
        set { _temperaturaCelsius = value; }
    }

    public double Fahrenheit
    {
        get { return _temperaturaCelsius * 9 / 5 + 32; }
        set { _temperaturaCelsius = (value - 32) * 5 / 9; }
    }

    public double Kelvin
    {
        get { return _temperaturaCelsius + 273.15; }
        set { _temperaturaCelsius = value - 273.15; }
    }
}


Program:
namespace Termometro
{
    internal class Program
    {
        static void Main(string[] args)
        {
            Termometro meuTermometro = new Termometro();

            meuTermometro.Celsius = 25;

            Console.WriteLine("- Definindo temperatura em Celsius: 25°C -");
            Console.WriteLine($"Celsius:    {meuTermometro.Celsius:F2} °C");
            Console.WriteLine($"Fahrenheit: {meuTermometro.Fahrenheit:F2} °F");
            Console.WriteLine($"Kelvin:     {meuTermometro.Kelvin:F2} K");

            Console.WriteLine();

            meuTermometro.Fahrenheit = 98.6;

            Console.WriteLine("- Definindo temperatura em Fahrenheit: 98.6°F -");
            Console.WriteLine($"Celsius:    {meuTermometro.Celsius:F2} °C");
            Console.WriteLine($"Fahrenheit: {meuTermometro.Fahrenheit:F2} °F");
            Console.WriteLine($"Kelvin:     {meuTermometro.Kelvin:F2} K");

            Console.WriteLine();

            meuTermometro.Kelvin = 300;

            Console.WriteLine("- Definindo temperatura em Kelvin: 300 K -");
            Console.WriteLine($"Celsius:    {meuTermometro.Celsius:F2} °C");
            Console.WriteLine($"Fahrenheit: {meuTermometro.Fahrenheit:F2} °F");
            Console.WriteLine($"Kelvin:     {meuTermometro.Kelvin:F2} K");
        }
    }
}
